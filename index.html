function doPost(e) { try { // 현재 연결된 스프레드시트의 첫 번째 시트 가져오기 var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; // 웹 앱에서 보낸 데이터 파싱 var data = JSON.parse(e.postData.contents); // 1행 헤더가 기본 4열(A~D)만 있다면 E~J열 헤더 자동 추가 if (sheet.getLastColumn() <= 4) { sheet.getRange(1, 5, 1, 6).setValues([[ "제출 시각", "탭1_하루의일기", "탭2_반어표현", "탭3_역설표현", "탭4_반어이유", "탭5_역설이유" ]]); } // 스프레드시트 맨 아래 행에 학생 데이터 기록 (선생님만 열람 가능) sheet.appendRow([ data.name, // A열: 이름 data.studentId, // B열: 학번 data.score, // C열: 퀴즈 점수 (100점 만점 기준 검증 통과 점수) data.passed, // D열: 통과 여부 ("통과") data.submittedAt, // E열: 제출 시각 data.tab1, // F열: 하루의 일기 data.tab2, // G열: 반어 표현 data.tab3, // H열: 역설 표현 data.tab4, // I열: 반어 선택 이유 data.tab5 // J열: 역설 선택 이유 ]); return ContentService .createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ result: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
